🔍 Bricks Save Detection Diagnostics

Run this in your browser console on a Bricks editor page

Step 1: Check Script Loading

Step 2: Check Bricks Elements

Step 3: Check Vue Store Access

Step 4: Manual Save Watcher

⚠️ Copy & Run in Browser Console

Copy the code below and paste it into your browser console while on a Bricks editor page:

// Bricks Save Detection Diagnostic
(function() {
    console.log('=== WINDEN BRICKS SAVE DIAGNOSTIC ===');

    // 1. Check if post-save-compile.js is loaded
    const scripts = Array.from(document.scripts);
    const postSaveScript = scripts.find(s => s.src.includes('post-save-compile'));
    console.log('1. post-save-compile.js loaded:', !!postSaveScript);
    if (postSaveScript) {
        console.log('   Script URL:', postSaveScript.src);
    }

    // 2. Check Bricks elements
    const panel = document.getElementById('bricks-panel-controls');
    const iframe = document.getElementById('bricks-builder-iframe');
    console.log('2. Bricks panel exists:', !!panel);
    console.log('   Bricks iframe exists:', !!iframe);

    // 3. Check Vue instance
    if (panel) {
        console.log('3. Panel has _vnode:', !!panel._vnode);
        if (panel._vnode) {
            const instance = panel._vnode.component?.proxy;
            console.log('   Vue instance:', !!instance);
            if (instance) {
                const store = instance.$_state;
                console.log('   Vue store ($_state):', !!store);
                if (store) {
                    console.log('   isSaving property:', 'isSaving' in store);
                    console.log('   Current isSaving:', store.isSaving);
                }
            }
        }
    }

    // 4. Setup manual watcher
    if (panel?._vnode?.component?.proxy) {
        const instance = panel._vnode.component.proxy;
        const store = instance.$_state;

        console.log('4. Setting up manual save watcher...');
        instance.$watch(
            () => store.isSaving,
            (isSaving, wasSaving) => {
                console.log('🔔 SAVE STATE CHANGED:', {
                    from: wasSaving,
                    to: isSaving,
                    timestamp: new Date().toLocaleTimeString()
                });

                if (wasSaving && !isSaving) {
                    console.log('✅ SAVE COMPLETED!');
                }
                if (!wasSaving && isSaving) {
                    console.log('⏳ SAVE STARTED...');
                }
            }
        );
        console.log('✅ Manual watcher installed! Click Save in Bricks to test.');
    } else {
        console.log('❌ Cannot setup watcher - Vue instance not accessible');
    }

    console.log('=== END DIAGNOSTIC ===');
})();
